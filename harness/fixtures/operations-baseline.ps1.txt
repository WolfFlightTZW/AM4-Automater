function Write-AtomicJson([string]$path,[object]$obj) {
 $temp=Join-Path (Split-Path -Parent $path) ('.'+[IO.Path]::GetFileName($path)+'.'+[guid]::NewGuid().ToString('N')+'.tmp')
 try {
  $json=ConvertTo-Json -InputObject $obj -Depth 18
  [IO.File]::WriteAllText($temp,$json,(New-Object System.Text.UTF8Encoding($false)))
  if(Test-Path -LiteralPath $path){
   # Windows PowerShell 5.1/.NET Framework requires a legal backup path for File.Replace.
   # Keep replacement on the same volume for atomicity; discard backup after success.
   $backup=$temp+'.bak'
   [IO.File]::Replace($temp,$path,$backup)
   if(Test-Path -LiteralPath $backup){Remove-Item -LiteralPath $backup -Force}
  }else{[IO.File]::Move($temp,$path)}
 }finally{if(Test-Path -LiteralPath $temp){Remove-Item -LiteralPath $temp -Force}}
}
